Class Doctor
    Attributes:
        name
        specialization
        experienceYears

    Constructor(name, specialization, experienceYears):
        this.name = name
        this.specialization = specialization
        this.experienceYears = experienceYears

    Method getDetails():
        RETURN name + " - " + specialization + " (" + experienceYears + " years)"

Class HospitalDepartmentDirectory
    Attributes:
        departmentMap ← Map<String, List<Doctor>>

    Methods:
        addDoctor(departmentName, doctor):
            If departmentName not in departmentMap:
                departmentMap[departmentName] = empty list
            Append doctor to departmentMap[departmentName]

        getDoctorsByDepartment(departmentName):
            If departmentName in departmentMap:
                For each doctor in departmentMap[departmentName]:
                    Print doctor.getDetails()
            Else:
                Print "No such department exists."

Main Program:
    directory = new HospitalDepartmentDirectory()

    directory.addDoctor("Cardiology", new Doctor("Dr. Mehta", "Cardiologist", 15))
    directory.addDoctor("Cardiology", new Doctor("Dr. Kapoor", "Cardiologist", 10))
    directory.addDoctor("Neurology", new Doctor("Dr. Sharma", "Neurologist", 10))
    directory.addDoctor("Neurology", new Doctor("Dr. Das", "Neurologist", 8))
    directory.addDoctor("Pediatrics", new Doctor("Dr. Ritu", "Pediatrician", 12))
    directory.addDoctor("Pediatrics", new Doctor("Dr. Anjali", "Pediatrician", 7))

    PRINT "Doctors in Neurology:"
    directory.getDoctorsByDepartment("Neurology")
